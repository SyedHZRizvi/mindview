"""Three-question quiz.""" def ask(question, answer): given = input(question + " ").strip().lower() return given == answer.lower() QUESTIONS = [ ("Capital of Canada?", "Ottawa"), ("2 + 2 * 3 = ?", "8"), ("Author of the first algorithm?", "Lovelace"), ] score = 0 for q, a in QUESTIONS: if ask(q, a): score += 1 percent = score / len(QUESTIONS) * 100 print(f"{score} / {len(QUESTIONS)} = {percent:.0f}%")
